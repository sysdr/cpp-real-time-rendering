cmake_minimum_required(VERSION 3.15)
project(OpenGLContextDemo CXX)

if(DEFINED ENV{VCPKG_ROOT} AND NOT CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE FILEPATH "Vcpkg toolchain file")
endif()
if(NOT CMAKE_TOOLCHAIN_FILE)
    find_program(VCPKG_ROOT_PATH vcpkg)
    if(VCPKG_ROOT_PATH)
        get_filename_component(VCPKG_ROOT_PATH ${VCPKG_ROOT_PATH} DIRECTORY)
        set(CMAKE_TOOLCHAIN_FILE "${VCPKG_ROOT_PATH}/scripts/buildsystems/vcpkg.cmake" CACHE FILEPATH "Vcpkg toolchain file")
    else()
        message(FATAL_ERROR "vcpkg not found. Set VCPKG_ROOT or add vcpkg to PATH.")
    endif()
endif()

find_package(glfw3 CONFIG REQUIRED)
find_package(glad CONFIG REQUIRED)
add_executable(OpenGLContextDemo src/main.cpp)
target_link_libraries(OpenGLContextDemo PRIVATE glfw glad::glad)
set_target_properties(OpenGLContextDemo PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)
